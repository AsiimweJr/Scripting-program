def compile_student_marks(names_list):
    subjects = ['Data ANALYSIS', 'Math For Eng', 'Molecular Biology', 'Phylogenetics']
    class_data = {}
    
    for name in names_list:
        student_marks = {}
        for subject in subjects:
            student_marks[subject] = random.randint(50, 100)
            
        total_marks = sum(student_marks.values())
        average_mark = total_marks / len(subjects)
        
        grade = 'A' if average_mark >= 80 else 'B' if average_mark >= 65 else 'C' if average_mark >= 50 else 'F'
            
        class_data[name] = {
            "marks": student_marks, "total": total_marks, "average": round(average_mark, 2), "grade": grade
        }
    return class_data
